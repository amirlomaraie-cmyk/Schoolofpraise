        function openSubjectPage(subjectKey) {
            const sub = subjectsData[subjectKey];
            if (!sub) return;

            const headerEl = document.getElementById('subject-header');
            headerEl.className = `p-8 rounded-2xl shadow-lg text-white mb-8 flex items-center gap-6 ${sub.bgColor} border border-coptic-gold/40`;
            
            let iconHtml = sub.customLogo ? 
                `<div class="w-16 h-16 bg-coptic-gold/20 rounded-2xl flex items-center justify-center font-black text-4xl shadow-inner border border-coptic-gold text-coptic-gold">${sub.customLogo}</div>` :
                `<div class="w-16 h-16 bg-coptic-gold/20 rounded-2xl flex items-center justify-center text-3xl shadow-inner border border-coptic-gold text-coptic-gold"><i class="fa-solid ${sub.icon}"></i></div>`;

            headerEl.innerHTML = `${iconHtml}<div><h3 class="text-3xl font-extrabold mb-1 text-coptic-gold">${sub.title}</h3><p class="text-slate-200 text-sm">${sub.desc}</p></div>`;

            const listEl = document.getElementById('subject-classes-list');
            listEl.innerHTML = sub.classes.map(cls => `
                <div class="flex items-center justify-between p-4 bg-slate-50 border border-slate-200 rounded-xl hover:bg-slate-100 transition">
                    <span class="font-bold text-slate-700">${cls.name}</span>
                    <button onclick="openReader('${cls.name} - ${sub.title}', '${cls.link}')" class="bg-coptic-navy text-coptic-gold font-bold px-4 py-2 rounded-lg hover:bg-slate-800 transition text-sm border border-coptic-gold/30">قراءة المنهج</button>
                </div>
            `).join('');

            showPage('subject-details');
        }

        function renderBooks() {
            const grid = document.getElementById('books-grid');
            if (!grid) return;
            
            grid.innerHTML = booksList.map(book => {
                let buttonsHtml = '';
                let arLink = book.copticArLink.replace('/view?usp=drive_link', '/preview').replace('/view', '/preview');
                let coLink = book.copticLink.replace('/view?usp=drive_link', '/preview').replace('/view', '/preview');

                if (book.copticArLink) {
                    buttonsHtml += `<button onclick="openReader('${book.name} (قبطي مُعرَّب)', '${arLink}')" class="w-full bg-coptic-navy text-coptic-gold font-bold py-2 px-3 rounded-lg hover:bg-slate-800 transition text-xs mb-2 border border-coptic-gold/30">قبطي مُعرَّب</button>`;
                }
                if (book.copticLink) {
                    buttonsHtml += `<button onclick="openReader('${book.name} (قبطي قبطي)', '${coLink}')" class="w-full bg-coptic-gold text-coptic-navy font-bold py-2 px-3 rounded-lg hover:bg-coptic-darkGold hover:text-white transition text-xs">قبطي قبطي</button>`;
                }
                if (!book.copticArLink && !book.copticLink) {
                    buttonsHtml += `<span class="text-xs text-slate-400 italic">قريباً...</span>`;
                }

                return `
                    <div class="bg-white rounded-2xl shadow-md border border-slate-200 overflow-hidden flex flex-col justify-between hover:shadow-xl transition">
                        <div class="bg-gradient-to-r ${book.bgGradient} p-6 text-white text-center border-b-2 border-coptic-gold/40">
                            <div class="w-12 h-12 bg-coptic-gold/20 rounded-xl flex items-center justify-center mx-auto mb-3 text-2xl shadow-inner border border-coptic-gold/40 text-coptic-gold"><i class="fa-solid ${book.icon}"></i></div>
                            <h4 class="font-bold text-base leading-snug">${book.name}</h4>
                        </div>
                        <div class="p-4 bg-slate-50 border-t border-slate-100 text-center">${buttonsHtml}</div>
                    </div>
                `;
            }).join('');
        }

        function openReader(title, link) {
            document.getElementById('reader-title').textContent = title;
            document.getElementById('reader-frame').src = link;
            showPage('reader');
        }
